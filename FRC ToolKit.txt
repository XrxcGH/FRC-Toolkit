export function frcDefinition(definition is map) returns map
{
  var d = definition;
  if (d.partMaterial != undefined)
    d.material = d.partMaterial;
  if (d.partLength != undefined)
    d.length = d.partLength;
  if (d.beltLength != undefined)
    d.length = d.beltLength;
  if (d.cubeSize != undefined)
    d.size = d.cubeSize;
  if (d.drivePlane != undefined)
    d.plane = d.drivePlane;
  if (d.floorFace != undefined)
    d.floor = d.floorFace;
  if (d.flipSide != undefined)
    d.flip = d.flipSide;
  if (d.chamferSize != undefined)
    d.chamfer = d.chamferSize;
  return d;
}

export function frcGroupList(definition is map) returns array
{
  const q = [definition.bodies1, definition.bodies2,
       definition.bodies3, definition.bodies4];
  const m = [definition.material1, definition.material2,
       definition.material3, definition.material4];
  const on = [true, definition.group2, definition.group3, definition.group4];
  var out = [];
  for (var i = 0; i < 4; i += 1)
  {
    if (on[i] == true && q[i] is Query)
      out = append(out, { "bodies" : q[i], "material" : m[i] });
  }
  return out;
}
